window.__run = async () => {
  const control = build('control');
  const patched = build('patched');
  const controlInstant = build('control-instant', true);
  const patchedInstant = build('patched-instant', true);
  const patchedInstantAnchored = build('patched-instant-anchored', false);

  log(await scenario(control, 3, 'control animated (no plugin)'));
  log(await instantScenario(controlInstant, 3, 'control instant, no native anchoring (no plugin)'));

  const script = document.createElement('script');
  script.src = BUNDLE_URL;
  document.body.append(script);
  await new Promise((resolve, reject) => { script.onload = resolve; script.onerror = reject; });
  const registration = window.__loaded;
  if (registration === undefined) throw new Error('bundle did not register');
  const plugin = registration.factory(() => { throw new Error('external module requested'); });
  plugin.apply({ effect: (run) => { run(); return () => {}; } });

  log({ pluginName: plugin.name });
  log(await scenario(patched, 3, 'patched animated (plugin)'));
  log(await instantScenario(patchedInstant, 3, 'patched instant, no native anchoring (plugin)'));
  log(await instantScenario(patchedInstantAnchored, 3, 'patched instant, native anchoring on (plugin)'));
  return JSON.stringify(out, null, 1);
};
